// Send form data to Supabase signup or login endpoint function handleAuthSubmit() { var email = document.getElementById("auth-email").value.trim(); var password = document.getElementById("auth-password").value.trim(); if (!email || !password) { return alert("Fill out both input boxes!"); } var endpoint = isSignUpMode ? "/auth/v1/signup" : "/auth/v1/token?grant_type=password"; var xhr = new XMLHttpRequest(); xhr.open("POST", SUPABASE_URL + endpoint, true); xhr.setRequestHeader("apikey", SUPABASE_KEY); xhr.setRequestHeader("Content-Type", "application/json"); xhr.onreadystatechange = function() { if (xhr.readyState === 4) { if (xhr.status >= 200 && xhr.status <= 299) { var data = JSON.parse(xhr.responseText); if (isSignUpMode) { alert("Registration successful! You can now log in using these credentials."); toggleAuthMode(); } else { localStorage.setItem("sb_token", data.access_token); localStorage.setItem("sb_user_email", data.user.email); document.getElementById("auth-email").value = ""; document.getElementById("auth-password").value = ""; checkCurrentSession(); } } else { var err = JSON.parse(xhr.responseText || "{}"); alert("Authentication Error: " + (err.error_description || err.msg || xhr.status)); } } }; xhr.send(JSON.stringify({ "email": email, "password": password })); } function handleLogout() { localStorage.removeItem("sb_token"); localStorage.removeItem("sb_user_email"); checkCurrentSession(); } function fetchRequestMessages() { var feed = document.getElementById("chat-stream"); var xhr = new XMLHttpRequest(); xhr.open("GET", SUPABASE_URL + "/rest/v1/online_chat?select=*&order=created_at.asc&limit=15", true); xhr.setRequestHeader("apikey", SUPABASE_KEY); xhr.onreadystatechange = function() { if (xhr.readyState === 4 && xhr.status === 200) { var data = JSON.parse(xhr.responseText); var htmlBuffer = ""; if (data.length === 0) { feed.innerHTML = "
The board is empty.
"; return; } for (var i = 0; i < data.length; i++) { var timeStr = ""; try { var dateObj = new Date(data[i].created_at); timeStr = dateObj.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'}); } catch(e) { timeStr = "--:--"; } htmlBuffer += "
" + "" + timeStr + "" + "" + escapeHtml(data[i].name) + ": " + "" + escapeHtml(data[i].message) + "" + "
"; } feed.innerHTML = htmlBuffer; var isAtBottom = (feed.scrollHeight - feed.clientHeight - feed.scrollTop) < 40; if (isAtBottom) { feed.scrollTop = feed.scrollHeight; } } }; xhr.send(); } function submitRequestMessage() { var msgInput = document.getElementById("user-msg").value.trim(); var userEmail = localStorage.getItem("sb_user_email"); var jwtToken = localStorage.getItem("sb_token"); if (!msgInput) return alert("Write something first!"); var parsedUsername = userEmail.split('@')[0]; var xhr = new XMLHttpRequest(); xhr.open("POST", SUPABASE_URL + "/rest/v1/online_chat", true); xhr.setRequestHeader("apikey", SUPABASE_KEY); xhr.setRequestHeader("Authorization", "Bearer " + jwtToken); xhr.setRequestHeader("Content-Type", "application/json"); xhr.onreadystatechange = function() { if (xhr.readyState === 4) { if (xhr.status >= 200 && xhr.status <= 299) { document.getElementById("user-msg").value = ""; fetchRequestMessages(); } else { alert("Transmission failure. Status code: " + xhr.status); } } }; xhr.send(JSON.stringify({ "name": parsedUsername, "message": msgInput })); } function escapeHtml(str) { if (!str) return ""; return str.replace(/&/g, "&").replace(//g, ">"); } // Trigger state check on bootup checkCurrentSession();
